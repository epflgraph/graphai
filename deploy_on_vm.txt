# 0) Get repo
git clone <YOUR_REPO_URL> graphai
cd graphai

# 1) Prepare env
cp .env.deploy.example .env

GRAPH_IMAGE=epflgraph/graphai:latest
COMPOSE_PROFILES=cpu-fallback,local-infra
DB_HOST=mysql
ES_HOST=elasticsearch

# 2) Create local compose variant that mounts host models dir
cp docker-compose.deploy.yml docker-compose.local.yml
sed -i 's#graphai_models:/opt/models#./models:/opt/models#g' docker-compose.local.yml

# 3) Pull app image
docker pull epflgraph/graphai:latest

# 4) Download models to host ./models
chmod +x docker/setup-models.sh
IMAGE=epflgraph/graphai:latest MODEL_ROOT="$PWD/models" ./docker/setup-models.sh

# 5) Start full local stack
docker compose --env-file .env -f docker-compose.local.yml up -d

# 6) Check status/logs
docker compose --env-file .env -f docker-compose.local.yml ps
docker compose --env-file .env -f docker-compose.local.yml logs -f api

# 7) Stop
docker compose --env-file .env -f docker-compose.local.yml down
